  const sent = { run:{ run_id:'r', status:'ok', posts_fetched:1, data_as_of:'2025-06-01T00:00:00Z' }, snapshot:{ as_of:'2025-06-01T00:00:00.000Z', followers:5, following:null, posts:0 }, posts:[{ post_id:'a', created_at:'2025-06-01T00:00:00Z', url:null, body:'é 🚪', kind:'post' }], metrics:[{ post_id:'a', likes:0, reposts:null, replies:3, quotes:null, impressions:null }] };

test('Neon round-trip: write → readback hash match → NULL preserved → cleanup', async () => {
  if (!Store.configured()) return { skip:'no Neon connection configured' };
  const now = new Date();
  const ing = newIngest({ source_id:'selftest:' + uuid().slice(0,8), platform:'selftest', handle:'selftest', label:'Self-test (auto-deleted)' }, 'selftest', now);
  ing.run.data_as_of = now.toISOString(); ing.run.coverage_from = new Date(+now - DAY).toISOString(); ing.run.coverage_to = now.toISOString();
  ing.snapshot = { as_of:now.toISOString(), followers:2147483648, following:0, posts:null };
  ing.posts = [{ post_id:'a', created_at:now.toISOString(), url:'https://example.invalid/a', body:'Ünïcødé 🚪 "quotes" \\ back\nnewline\u0000nul', kind:'post' },
               { post_id:'b', created_at:null, url:null, body:'', kind:'reply' }];
  ing.metrics = [{ post_id:'a', likes:0, reposts:null, replies:3, quotes:null, impressions:9007199254 }, { post_id:'b', likes:null, reposts:null, replies:null, quotes:null, impressions:null }];
  await finalize(ing, now);
  try {
    await Store.writeIngest(ing);
    const rb = await Store.readback(ing);
    assert(rb.ok, 'readback mismatch: ' + rb.diff.join(' | '));
    const rows = await Store.q(`SELECT likes, reposts FROM ${SCHEMA}.post_metrics WHERE run_id=$1 AND post_id='a'`, [ing.run.run_id]);
    eq(n(rows[0].likes), 0, 'zero stays zero'); eq(rows[0].reposts, null, 'NULL stays NULL');
  } finally { await Store.q(`DELETE FROM ${SCHEMA}.sources WHERE source_id=$1`, [ing.source.source_id]); }
  const left = await Store.q(`SELECT count(*) AS c FROM ${SCHEMA}.ingest_runs WHERE run_id=$1`, [ing.run.run_id]);
  eq(n(left[0].c), 0, 'cleanup');
}, { integration:true });
test('Real private readback: latest run of every real source was read back and matched', async () => {
  if (!Store.configured()) return { skip:'no Neon connection configured' };
  const rows = await Store.q(`SELECT DISTINCT ON (r.source_id) r.source_id, s.platform, r.status, r.readback_ok, r.run_id, r.posts_fetched, ${I('r.readback_at')} AS readback_at
    FROM ${SCHEMA}.ingest_runs r JOIN ${SCHEMA}.sources s USING (source_id) WHERE s.platform <> 'selftest' ORDER BY r.source_id, r.started_at DESC`);
  if (!rows.length) return { skip:'no real collections in the store yet. Collect sources first' };
  const bad = rows.filter(r => !bool(r.readback_ok));
  assert(!bad.length, 'not verified: ' + bad.map(r => `${r.source_id} (readback_ok=${r.readback_ok ?? 'null'})`).join(', '));
  const missing = ['bluesky','mastodon','x_archive','typefully'].filter(p => !rows.some(r => r.platform === p));
  return { note:`${rows.length} source(s) verified: ` + rows.map(r => `${r.source_id} [${r.status}, ${r.posts_fetched} posts]`).join('; ') + (missing.length ? ` · never collected: ${missing.join(', ')}` : '') };
}, { integration:true });

async function runTests(which){
  const list = TESTS.filter(t => which === 'integration' ? t.integration : !t.integration);
  const ul = $('#tests'); ul.innerHTML = `
  • running ${list.length} ${which} test(s)…
    • `;
  const out = [];
  for (const t of list) {
    try { const r = await t.fn(); out.push({ name:t.name, status:r?.skip ? 'skip' : 'pass', note:r?.skip || r?.note || '' }); }
    catch (e) { out.push({ name:t.name, status:'fail', note:e.message }); }
  }
  const c = s => out.filter(x => x.status === s).length;
  ul.innerHTML = `
  • ${which}: ${c('pass')} passed, ${c('fail')} failed, ${c('skip')} skipped
    • ` +
    out.map(x => `
  • ${x.status === 'pass' ? '✓' : x.status === 'fail' ? '✗' : '○'} ${esc(x.name)}${x.note ? ` (${esc(x.note)})` : ''}
    • `).join('');
  log(`${which} tests: ${c('pass')} passed, ${c('fail')} failed, ${c('skip')} skipped`, c('fail') ? 'bad' : c('skip') ? 'warn' : 'ok');
}
